      document.querySelectorAll('.page-section').forEach(sec => sec.classList.remove('active'));
      const target = document.querySelector(hash);
      if (target) {
        target.classList.add('active');
        window.scrollTo(0, 0);
      } else {
        document.querySelector('#home').classList.add('active');
