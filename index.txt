        function showTab(category, tabId, element) {
            var parentSection;
            if(category === 'vehicle') parentSection = document.getElementById('vehicles');
            else if(category === 'region') parentSection = document.getElementById('regions');
            else if(category === 'use') parentSection = document.getElementById('uses');

            if (!parentSection) return;

            var buttons = parentSection.querySelectorAll('.tab-btn');
            buttons.forEach(function(btn) {
                btn.classList.remove('active');
            });
            element.classList.add('active');

            var cards = parentSection.querySelectorAll('.vehicle-card, .service-card');
            cards.forEach(function(card) {
                card.classList.remove('active');
            });
            
            var targetCard = document.getElementById(tabId);
            if (targetCard) {
                targetCard.classList.add('active');
            }
        }
    </script>
